# YAMAHA RTX830 初期設定
# 回線: NTT フレッツ光
# プロバイダー: アンドライン
# 接続方式: IPoE (v6プラス MAP-E) ── IPv4/IPv6 デュアルスタック
#
# 【v6プラスの仕組み】
#   IPv6: LAN2 で DHCPv6-PD → IPoE ネイティブ接続
#   IPv4: MAP-E トンネル経由（PPPoEは不要）
#
# 【接続構成】
#   ONU ──(LAN2/WAN)── RTX830 ──(LAN1)── PC / スイッチ等
#
# 【適用方法】
#   1. RTX830 の管理画面 (http://192.168.11.1) または SSH/Telnet でログイン
#   2. 以下を上から順に貼り付けて実行
#   3. 最後に save を実行して保存
# -----------------------------------------------------------------------

# -----------------------------------------------------------------------
# システム基本設定
# -----------------------------------------------------------------------
console character utf-8
ip filter source-route on
ip filter directed-broadcast on

# -----------------------------------------------------------------------
# LAN1 設定（内部ネットワーク 192.168.11.0/24）
# -----------------------------------------------------------------------
ip lan1 address 192.168.11.1/24

# -----------------------------------------------------------------------
# LAN2 設定（WAN側 - IPoE IPv6ネイティブ）
#   DHCPv6-PD でグローバル IPv6 プレフィックスを取得する
# -----------------------------------------------------------------------
ipv6 lan2 dhcp service client ir=on

# -----------------------------------------------------------------------
# IPv6 LAN1 設定（RA で PC 側へアドレスを配布）
# -----------------------------------------------------------------------
ipv6 prefix 1 dhcp-prefix@lan2::/64
ipv6 lan1 address dhcp-prefix@lan2::1/64
ipv6 lan1 rtadv send 1 o_flag=on
ipv6 lan1 dhcp service server

# IPv6 デフォルトルート（IPoE 経由）
ipv6 route default gateway dhcp lan2

# -----------------------------------------------------------------------
# MAP-E トンネル設定（v6プラス IPv4通信用）
#   RTX830 が v6プラスのルールを自動適用するため
#   ユーザーによるパラメーター入力は不要
# -----------------------------------------------------------------------
tunnel select 1
 description tunnel v6plus-map-e
 tunnel encapsulation map-e
 tunnel map-e type v6plus
 ip tunnel mtu 1460
 ip tunnel tcp-mss-limit 1420
 ip tunnel nat descriptor 1 2 3 4 5
 tunnel enable 1

# IPv4 デフォルトルート（MAP-E トンネル経由）
# keepalive 1（IPv4）: 3秒間隔・3回失敗（9秒）でダウン判定 → ルート自動無効化・復旧
# keepalive 2（IPv6）: 同条件で IPv6 疎通を独立監視
ip keepalive 1 icmp-echo 3 3 8.8.8.8
ip keepalive 2 icmp-echo 3 3 2001:4860:4860::8888
ip route default gateway tunnel 1 keepalive 1

# -----------------------------------------------------------------------
# NAT / IPマスカレード（MAP-E 用）
#   v6プラスではポート番号が自動的に制限される（PSID方式）
#   outer アドレスは MAP-E ルールから自動決定されるため指定不要
# -----------------------------------------------------------------------

# descriptor 1: 現用サブネット（192.168.11.0/24）
nat descriptor type 1 masquerade
nat descriptor address inner 1 192.168.11.0/24

# descriptor 2〜5: VLAN1〜4 展開時用（将来の拡張向け先行設定）
# 【要変更：各VLANのサブネットが確定したら inner アドレスを修正すること】
nat descriptor type 2 masquerade
nat descriptor address inner 2 192.168.1.0/24

nat descriptor type 3 masquerade
nat descriptor address inner 3 192.168.2.0/24

nat descriptor type 4 masquerade
nat descriptor address inner 4 192.168.3.0/24

nat descriptor type 5 masquerade
nat descriptor address inner 5 192.168.4.0/24

# -----------------------------------------------------------------------
# パケットフィルター IPv4
# -----------------------------------------------------------------------
ip filter 100 reject * * udp,tcp 135 *
ip filter 101 reject * * udp,tcp * 135
ip filter 102 reject * * udp,tcp netbios_ns-netbios_ssn *
ip filter 103 reject * * udp,tcp * netbios_ns-netbios_ssn
ip filter 104 reject * * udp,tcp 445 *
ip filter 105 reject * * udp,tcp * 445
ip filter 106 reject 172.16.0.0/12 *
ip filter 107 reject 192.168.0.0/16 *
ip filter 108 reject 10.0.0.0/8 *
ip filter 200 pass * *

ip tunnel secure filter in 100 101 102 103 104 105 106 107 108 200
ip tunnel secure filter out 200

# -----------------------------------------------------------------------
# パケットフィルター IPv6
# -----------------------------------------------------------------------

# 静的フィルタ定義
ipv6 filter 100 pass * * icmp6 * *
ipv6 filter 102 pass * * udp * 546
ipv6 filter 103 pass * * 4 * *
ipv6 filter 200 reject * *

# ステートフルフィルタ定義（内側から始めた通信の戻りパケットを自動許可）
ipv6 filter dynamic 101 * * tcp
ipv6 filter dynamic 102 * * udp

# lan2（WAN側）: 静的許可 + ステートフル。不正着信は 200 で遮断
ipv6 lan2 secure filter in 100 102 103 200 dynamic 101 102
# lan2 送信: 全許可（RSやDHCPv6 Solicitをブロックしないようにする）
ipv6 filter 201 pass * *
ipv6 lan2 secure filter out 201
# lan1（内側）: 内部通信は全許可
ipv6 lan1 secure filter in 201
ipv6 lan1 secure filter out 201

# -----------------------------------------------------------------------
# DHCP サーバー
#   スコープ全体: 192.168.11.2 〜 192.168.11.191
#   固定割り当て（bind）: .18 / .28 / .40 は MAC で予約済み
#   動的配布: .50 〜 .191（bindで予約した以外のアドレス）
# -----------------------------------------------------------------------
dhcp service server
dhcp server rfc2131 compliant except remain-silent
dhcp scope 1 192.168.11.2-192.168.11.191/24

# MACアドレスによる固定IP割り当て
dhcp scope bind 1 192.168.11.18 ethernet b8:27:eb:ba:0b:00
dhcp scope bind 1 192.168.11.28 ethernet 88:a2:9e:88:00:14
dhcp scope bind 1 192.168.11.40 ethernet c4:3c:ea:2b:63:90

# -----------------------------------------------------------------------
# DNS（IPoE 経由でプロバイダーのDNSを自動取得）
# -----------------------------------------------------------------------
dns service recursive
dns server dhcp lan2
dns private address spoof on

# -----------------------------------------------------------------------
# NTP（時刻同期）
# -----------------------------------------------------------------------
schedule at 1 startup * ntpdate ntp.nict.jp
schedule at 2 */* 04:00 * ntpdate ntp.nict.jp

# -----------------------------------------------------------------------
# IPoE セッション監視
# -----------------------------------------------------------------------
# keepalive 動作概要:
#   keepalive 1（IPv4）: 3秒間隔・3回失敗（9秒）でダウン判定 → デフォルトルート無効化
#   keepalive 2（IPv6）: 同条件で IPv6 疎通を独立監視
#   ※ keepalive 定義は MAP-E トンネルセクション内に記載済み
#
# IPv6(DHCPv6-PD) 定期更新: 毎日03:30に renew を送信
schedule at 3 */* 03:30 * dhcp renew client lan2
#
# セッション断が長引く場合の強制復旧: 毎日04:30にLAN2を再起動
# （通常は不要。頻繁に切れる場合のみコメントを外す）
# schedule at 4 */* 04:30 * lan restart lan2

# -----------------------------------------------------------------------
# ログ設定（SDカード保存・ローテーション）
# -----------------------------------------------------------------------
# notice レベル以上のイベント（トンネル断・keepalive失敗等）を記録
syslog notice on
syslog debug off
# SDカードへの保存を有効化（1ファイル最大1MB × 4世代ローテーション）
# 【要確認：RTX830でSDカードへの出力先が内部フラッシュになる場合は
#   メーカーマニュアルで `syslog save` の動作を確認すること】
syslog save on
syslog save file-size 1024
syslog save file-count 4

# -----------------------------------------------------------------------
# 設定保存（必ず最後に実行）
# -----------------------------------------------------------------------
# save
