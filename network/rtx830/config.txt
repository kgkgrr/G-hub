# YAMAHA RTX830 初期設定
# 回線: NTT フレッツ光
# プロバイダー: アンドライン (PPPoE接続、IPv4/IPv6 デュアルスタック)
#
# 【事前準備】
#   アンドラインから届いた書類に記載の以下を手元に用意してください:
#     - PPPoE接続ID  例: xxxxxx@andline.ne.jp
#     - PPPoEパスワード
#
# 【接続構成】
#   ONU (NTT光終端装置)
#     └─ LAN2 (WAN側) ─ RTX830 ─ LAN1 (内部LAN) ─ PC / スイッチ等
#
# 【適用方法】
#   1. RTX830 の管理画面 (http://192.168.1.1) または Telnet/SSH でログイン
#   2. 以下の設定を上から順に貼り付けて実行
#   3. save コマンドで設定を保存
# -----------------------------------------------------------------------

# -----------------------------------------------------------------------
# システム基本設定
# -----------------------------------------------------------------------
console character utf-8
ip route default gateway pp 1
ip filter source-route on
ip filter directed-broadcast on

# -----------------------------------------------------------------------
# LAN1 設定（内部ネットワーク 192.168.1.0/24）
# -----------------------------------------------------------------------
ip lan1 address 192.168.1.1/24

# -----------------------------------------------------------------------
# PPPoE 設定（アンドライン）
#   ★ pp auth myname の [USER] と [PASS] を実際の値に書き換えてください
# -----------------------------------------------------------------------
pp select 1
 description pp andline-pppoe
 pp keepalive interval 30 retry-interval=30 count=12
 pp always-on on
 pppoe use lan2
 pppoe auto disconnect off
 pp auth accept pap chap
 pp auth myname [USER@andline.ne.jp] [PASS]
 ppp lcp mtu on 1454
 ppp ipcp ipaddress on
 ppp ipcp msext on
 ppp ccp type none
 ip pp address dhcp
 ip pp nat descriptor 1
 ipv6 pp dhcp service client ir=off
 ipv6 pp address dhcp-prefix@lan1::
 pp enable 1

# -----------------------------------------------------------------------
# NAT / IPマスカレード
# -----------------------------------------------------------------------
nat descriptor type 1 masquerade
nat descriptor address outer 1 ipcp
nat descriptor address inner 1 192.168.1.0/24

# -----------------------------------------------------------------------
# IPv6 LAN1 設定（RA でPC側へプレフィックスを配布）
# -----------------------------------------------------------------------
ipv6 prefix 1 dhcp-prefix@lan1::/64
ipv6 lan1 address dhcp-prefix@lan1::1/64
ipv6 lan1 rtadv send 1 o_flag=on
ipv6 lan1 dhcp service server

# IPv6 デフォルトルート
ipv6 route default gateway pp 1

# -----------------------------------------------------------------------
# パケットフィルター IPv4（外部からの不正アクセス防止）
# -----------------------------------------------------------------------
ip filter 100 reject * * udp,tcp 135 *
ip filter 101 reject * * udp,tcp * 135
ip filter 102 reject * * udp,tcp netbios_ns-netbios_ssn *
ip filter 103 reject * * udp,tcp * netbios_ns-netbios_ssn
ip filter 104 reject * * udp,tcp 445 *
ip filter 105 reject * * udp,tcp * 445
ip filter 106 reject 172.16.0.0/12 *
ip filter 107 reject 192.168.0.0/16 *
ip filter 108 reject 10.0.0.0/8 *
ip filter 200 pass * *

ip pp secure filter in 100 101 102 103 104 105 106 107 108 200
ip pp secure filter out 200

# -----------------------------------------------------------------------
# パケットフィルター IPv6（外部からの不正アクセス防止）
# -----------------------------------------------------------------------
ipv6 filter 100 pass * * icmp6 * *
ipv6 filter 101 pass * * tcp * established
ipv6 filter 102 pass * * udp * 546
ipv6 filter 200 reject * *

ipv6 pp secure filter in 100 101 102 200
ipv6 pp secure filter out 200
ipv6 lan1 secure filter in 200
ipv6 lan1 secure filter out 200

# -----------------------------------------------------------------------
# DHCP サーバー（PC等に自動でIPを配布）
#   配布範囲: 192.168.1.2 〜 192.168.1.191
# -----------------------------------------------------------------------
dhcp service server
dhcp server rfc2131 compliant except remain-silent
dhcp scope 1 192.168.1.2-192.168.1.191/24

# -----------------------------------------------------------------------
# DNS（プロバイダーのDNSを自動取得）
# -----------------------------------------------------------------------
dns service recursive
dns server pp 1
dns private address spoof on

# -----------------------------------------------------------------------
# NTP（時刻同期）
# -----------------------------------------------------------------------
schedule at 1 startup * ntpdate ntp.nict.jp
schedule at 2 */* 04:00 * ntpdate ntp.nict.jp

# -----------------------------------------------------------------------
# 設定保存（必ず最後に実行）
# -----------------------------------------------------------------------
# save
